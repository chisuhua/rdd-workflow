```python
    """Build deps_data with `blocks` field: [from, to] pairs."""

    return {"blocks": blocks}

    """A blocks B,C; B,C block D → waves = [[A], [B,C], [D]]."""
    out = compute_waves(_deps([("a", "b"), ("a", "c"), ("b", "d"), ("c", "d")]))
    assert out == [["a"], ["b", "c"], ["d"]]

    """Independent changes all fit in wave 0."""
    out = compute_waves(_deps([("a", "b")]))

    # Only entries with NO inbound edges AND no required position land in wave 0.

    """A→B→A cycle → WaveCycleError."""
    with pytest.raises(WaveCycleError):
        compute_waves(_deps([("a", "b"), ("b", "a")]))
```

```python
"""Wave DAG computation from deps-analysis.json.

Per phase-2-general-20260829063801:
- compute_waves(deps_data) → list[list[str]]
  - wave 0: nodes with no inbound edges (can start immediately)
  - wave N+1: nodes whose ALL inbound neighbors are in waves 0..N
  - raises WaveCycleError if a cycle is detected
"""
from __future__ import annotations
import heapq
from typing import Dict, List, Set


class WaveCycleError(Exception):
    """Raised when deps form a cycle and wave computation cannot proceed."""


def _index_blocks(deps_data: dict) -> Dict[str, Set[str]]:
    """Return {node: set(downstream nodes it blocks)} from deps_data.blocks."""
    out: Dict[str, Set[str]] = {}
    for entry in deps_data.get("blocks", []):
        src = entry["from"]
        dst = entry["to"]
        out.setdefault(src, set()).add(dst)
        out.setdefault(dst, set())
    return out


def _inbound(blocks: Dict[str, Set[str]]) -> Dict[str, Set[str]]:
    """Reverse map: {node: set(upstream nodes that block it)}."""
    out: Dict[str, Set[str]] = {n: set() for n in blocks}
    for src, dsts in blocks.items():
        for dst in dsts:
            out[dst].add(src)
    return out


def compute_waves(deps_data: dict) -> List[List[str]]:
    """Kahn-style level-sorted wave partition. Ties broken by node name
    for deterministic output (tests can assert exact waves)."""
    blocks = _index_blocks(deps_data)
    if not blocks:
        return []

    inbound = _inbound(blocks)
    remaining_in = {n: set(s) for n, s in inbound.items()}
    waves: List[List[str]] = []
    processed: Set[str] = set()

    ready = sorted([n for n, s in remaining_in.items() if not s])
    while ready:
        waves.append(ready)
        for n in ready:
            processed.add(n)
        next_ready: list = []
        for n in ready:
            for downstream in blocks.get(n, set()):
                remaining_in[downstream].discard(n)
                if downstream not in processed and not remaining_in[downstream]:
                    next_ready.append(downstream)
        ready = sorted(set(next_ready) - processed)

    if processed != set(blocks):
        unprocessed = set(blocks) - processed
        raise WaveCycleError(f"Cycle detected; unprocessed: {sorted(unprocessed)}")
    return waves
```

```python
from __future__ import annotations
import time

from _lib.wave_compute import compute_waves


def _gen_chain(n):
    """n nodes in linear chain: 0→1→2→...→n-1."""
    return {"blocks": [{"from": str(i), "to": str(i + 1)} for i in range(n - 1)]}


def test_compute_waves_perf_5_changes_50_tasks():
    """5 changes × 10 tasks each → ~50 block edges → must finish < 100ms."""
    # 5 changes named c0..c4, each with 10 tasks (t_cN_tM); c0→c1→c2→c3→c4

    t0 = time.perf_counter()
    out = compute_waves(deps)
    elapsed_ms = (time.perf_counter() - t0) * 1000
    assert elapsed_ms < 100, f"compute_waves took {elapsed_ms:.1f}ms (limit 100ms)"
```
