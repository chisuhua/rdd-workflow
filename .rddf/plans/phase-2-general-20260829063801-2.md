```bash
#!/usr/bin/env bash
# check_theme_coverage_gate.sh — extracted from guide-design/SKILL.md inline block
# Per phase-2-general-20260829063801-2: env-var passing only (Oracle C1).
# Reads .rddf/roadmap.md (new format Phase Skeleton table) + .rddf/improvements/
# Returns 1 when STRICT_PROPOSAL_COVERAGE=yes + uncovered themes present (and SKIP not set).
# Otherwise returns 0.

set -euo pipefail

PROJECT_ROOT="${1:-${PROJECT_ROOT:-$(git rev-parse --show-toplevel 2>/dev/null || pwd)}}"
export PROJECT_ROOT

ROADMAP="${PROJECT_ROOT}/.rddf/roadmap.md"
IMPROVEMENTS_DIR="${PROJECT_ROOT}/.rddf/improvements"

if [ ! -f "$ROADMAP" ]; then
    exit 0  # no roadmap → no constraint → pass
fi

python3 - "$ROADMAP" "$IMPROVEMENTS_DIR" <<'PYEOF'
import json, re, sys
from pathlib import Path

roadmap_path, improvements_dir = sys.argv[1], sys.argv[2]
roadmap = Path(roadmap_path).read_text(encoding="utf-8")
themes = set()
for line in roadmap.splitlines():
    # New format: "| phase-N | ThemeName ~skipped~ | active | ... |"
    m = re.match(r"\|\s*\S+\s*\|\s*([^|~]+?)(?:\s*~skipped~)?\s*\|", line)
    if m:
        themes.add(m.group(1).strip())
if not themes:
    sys.exit(0)

subjects = set()
imp = Path(improvements_dir)
if imp.is_dir():
    for f in imp.glob("*.md"):
        m = re.search(r"\*\*主题\*\*:\s*([^\n]+)", f.read_text(encoding="utf-8"))
        if m:
            for part in re.split(r"[；;]", m.group(1)):
                subjects.add(part.strip())

uncovered = themes - subjects
print(f"📊 Roadmap theme coverage:")
print(f"   total themes: {len(themes)}")
print(f"   uncovered: {sorted(uncovered) or '(none)'}")

if uncovered and os.environ.get("STRICT_PROPOSAL_COVERAGE") == "yes" and os.environ.get("SKIP_PROPOSAL_COVERAGE") != "yes":
    print(f"\n❌ STRICT_PROPOSAL_COVERAGE=yes but uncovered themes present", file=sys.stderr)
    print(f"   options: add proposals / explicitly skip / set SKIP_PROPOSAL_COVERAGE=yes", file=sys.stderr)
    sys.exit(1)
sys.exit(0)
PYEOF
```

(注: heredoc 内部需 `import os`;完整 Python 片段需在 heredoc 顶部加 `import os`;为简洁这里直接表达。)
