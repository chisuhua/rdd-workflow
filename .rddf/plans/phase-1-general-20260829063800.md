```python
def _list_parallel_sessions() -> int:
    """``sessions list-parallel`` — group active sessions by workflow_group.

    Prints a table where same-group sessions are shown together, with a
    per-group "parallel" hint. Exits 0 on success, 1 on coordinator error.
    """
    try:
        coord = _get_coordinator()
    except ImportError as e:
        print(f"❌ sessions: failed to import RddfSessionCoordinator: {e}", file=sys.stderr)
        return 1

    try:
        sessions = coord.list_sessions()
    except Exception as e:
        print(f"❌ sessions list-parallel: {e}", file=sys.stderr)
        return 1

    if not sessions:
        print("📭 no sessions recorded")
        print('   create one via: skill_use("guide-arch"|"guide-design"|"guide-plan"|"guide-ship")')
        return 0

    # Group by workflow_group (None → 'independent')
    from collections import defaultdict
    groups: dict = defaultdict(list)
    for s in sessions:
        groups[s.workflow_group or "independent"].append(s)

    print("📊 Parallel session groups")
    for group_name, members in sorted(groups.items()):
        print(f"\n▸ workflow_group: {group_name}  ({len(members)} sessions)")
        print(f"{'SESSION_ID':<32} {'KIND':<14} {'STATE':<10} {'CHANGES':<8}")
        print(f"{'-'*32} {'-'*14} {'-'*10} {'-'*8}")
        for s in members:
            sid = (s.session_id or "?")[:32]
            kind = (s.kind or "?")[:14]
            state = (s.state or "?")[:10]
            n_changes = len(s.attached_changes or [])
            print(f"{sid:<32} {kind:<14} {state:<10} {n_changes:<8}")
    return 0
```

```python
"""tests/integration/test_rddf_session_crash_resume.py"""
"""SIGKILL crash → resume acceptance test.

Proves the "crash → resume" path from the proposal:
1. A session is created and left 'active'.
2. Simulate a crash: the owning process is SIGKILLed (no graceful close).
3. A new owner calls resume: transfers ownership, re-activates, works.
4. No leftover lock prevents the resume (LOCK_NB released on process death).
"""
from __future__ import annotations

import os
import signal
import subprocess
import sys
import time
from pathlib import Path

import pytest

from skills.rddf_session.scripts.rddf_session import (
    RddfSessionCoordinator,
    HeartbeatConfig,
    RddfSessionError,
)


def _write_probe_script(tmp_path: Path, sessions_file: str, session_id: str) -> Path:
    """Write a child script that creates a session then sleeps in a loop.

    The child is deliberately SIGKILLed to simulate a crash with no
    graceful close (no rbdd_session_hook_close).
    """
    script = tmp_path / "crash_child.py"
    script.write_text(f"""
import sys, time
sys.path.insert(0, "{tmp_path}")
sys.path.insert(0, ".")
from skills.rddf_session.scripts.rddf_session import (
    RddfSessionCoordinator, HeartbeatConfig,
)
c = RddfSessionCoordinator(
    "{sessions_file}",
    HeartbeatConfig(timeout_seconds=600, refresh_threshold_seconds=30),
)
sid = c.create_session(kind="stage_ship", owner_opencode_session_id="crash_owner")
print(sid, flush=True)
# Hold the flock: the store's with_file_lock uses LOCK_EX; keep the FD open
# so a subsequent acquisition would fail if the lock were not released on death.
import fcntl
with open("{sessions_file}.lock", "w") as lockf:
    fcntl.flock(lockf.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
    print("LOCK_HELD", flush=True)
    while True:
        time.sleep(1)
""")
    return script


def test_crash_then_resume(tmp_path):
    """A session whose owner was SIGKILLed can be resumed by a new owner."""
    sessions_file = str(tmp_path / "sessions.json")
    script = _write_probe_script(tmp_path, sessions_file, "N/A")

    # 1. Launch child that creates a session and holds a lock.
    proc = subprocess.Popen(
        [sys.executable, str(script)],
        stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True,
    )
    try:
        # Wait for the LOCK_HELD line.
        out_lines = []
        deadline = time.time() + 15
        held = False
        while time.time() < deadline:
            line = proc.stdout.readline()
            if line:
                out_lines.append(line.strip())
            if "LOCK_HELD" in out_lines:
                held = True
                break
            if proc.poll() is not None:
                break
        assert held, f"child never held lock; saw: {out_lines}"

        # 2. SIGKILL the child (crash, no graceful close).
        os.kill(proc.pid, signal.SIGKILL)
        proc.wait(timeout=10)
        assert proc.returncode == -9, "child must be killed by SIGKILL"

        # 3. New owner resumes: ownership transfer must succeed despite the
        #    crashed process's lock (kernel releases fcntl locks on exit).
        coord = RddfSessionCoordinator(
            sessions_file,
            HeartbeatConfig(timeout_seconds=60, refresh_threshold_seconds=30),
        )
        sessions = coord.list_sessions()
        assert len(sessions) == 1, "exactly 1 session must survive the crash"
        sid = sessions[0].session_id

        # Resume: transfer ownership + re-activate (idempotent).
        coord.transfer_ownership(sid, "new_owner")
        coord.update_session_status(sid, "active")

        resumed = coord.find_session(sid)
        assert resumed.owner_opencode_session_id == "new_owner"
        assert resumed.state == "active"

        # 4. A follow-up create_session (new work) still works — store not corrupted.
        sid2 = coord.create_session(kind="stage_plan", owner_opencode_session_id="new_owner")
        assert sid2 is not None
    finally:
        if proc.poll() is None:
            proc.kill()
            proc.wait()
```
